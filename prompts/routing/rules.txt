# These rules handle only high-confidence intent. Ambiguous web need is
# decided by the single classifier in prompts/routing/web-need.md.

[explicit-web]
browse\s+(?:the\s+)?web
search\s+(?:the\s+)?(?:web|internet|online)
look\s+.+?\s+up
use\s+the\s+(?:web|internet)
find\s+.+?\s+online

[explicit-project]
(?:my|this)\s+(?:project|repo|repository|codebase|application|app)

[current]
\b(?:currently|latest|newest|most\s+recent|right\s+now|live)\b
\bcurrent\s+(?:version|price|status|score|weather|forecast|schedule|availability|rate)\b
(?:what(?:'s|\s+is)|how(?:'s|\s+is)).{0,50}\b(?:weather|forecast)\b
\b(?:what\s+time\s+is\s+it|current\s+time|time\s+in\s+[a-z])
\b(?:open\s+now|opening\s+hours)\b
\b(?:current\s+score|live\s+score|what(?:'s|\s+is)\s+the\s+score)\b
\b(?:stock|share|bitcoin|crypto|exchange)\b.{0,30}\b(?:price|rate)\b
\b(?:flight|train|bus|ferry)\b.{0,30}\b(?:status|delay|schedule)\b
\bstill\b.{0,50}\b(?:available|open|operating|supported|active|legal|valid|down|up|in\s+stock)\b

[research]
\bresearch\b
\b(?:multiple|several|independent)\s+sources\b
\bsynthesize\s+(?:the\s+)?evidence\b

[debug]
\b(?:[a-z]*error|exception|traceback|bug|failed|failing|failure|unexpected|incorrect|troubleshoot)\b
\bsmallest\s+fix\b
\b(?:doesn't|does\s+not|won't|will\s+not|isn't|is\s+not)\s+(?:work|working|run|compile|start|load)\b
\b(?:returns?|outputs?|produces?)\b.{0,40}\b(?:undefined|null|wrong|incorrect|unexpected)\b
\bexpect(?:ed)?\b.{0,40}\b(?:but|instead)\b

[architecture]
\b(?:architecture|architectural|system\s+design|infrastructure\s+design)\b
\b(?:migration|deployment|caching)\s+strategy\b
\b(?:service\s+boundaries|message\s+queue)\b

[implement]
\b(?:add|modify|refactor|patch|remove|implement)\b
\b(?:write|create)\b.{0,80}\b(?:function|class|component|script|code|endpoint|test)\b

[writing]
\b(?:rewrite|rephrase|proofread|summarize|translate)\b
\bwrite\s+(?:a|an|the)?\s*(?:email|message|paragraph|poem|post|letter)\b

[explain-code]
\b(?:explain|walk\s+through|how\s+does|how\s+do)\b.{0,80}\b(?:function|class|interface|component|method|callback|code|implementation|source\s+file)\b
